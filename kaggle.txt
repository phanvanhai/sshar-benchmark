# ===== Cell 1: lấy code =====
import os, subprocess
os.environ["PYDEVD_DISABLE_FILE_VALIDATION"] = "1"
os.chdir("/kaggle/working")

REPO_URL, REPO = "https://github.com/phanvanhai/sshar-benchmark.git", "sshar-benchmark"
BRANCH = "v2"                 # nhánh code v2 (master = v1)
if not os.path.exists(REPO):
    subprocess.run(f"git clone -b {BRANCH} {REPO_URL} {REPO}", shell=True, check=True)
else:
    subprocess.run(f"git fetch origin && git checkout {BRANCH} && git pull origin {BRANCH}",
                   shell=True, check=True, cwd=REPO)
os.chdir(REPO)
print(subprocess.run("git branch --show-current && git log --oneline -1",
                     shell=True, capture_output=True, text=True).stdout)   # phải thấy: v2

# ===== Cell 3: cấu hình =====
ROOM2_ROOT  = "/kaggle/input/datasets/phanvanhai/cmr-har-room2" # <-- sửa: thư mục chứa room_02 (thí nghiệm 1-3)
XRF55_ROOT  = "/kaggle/input/datasets/phanvanhai/xrf55-8act/XRF55_8act_dataset" # <-- sửa: thư mục chứa file .npy XRF55 (thí nghiệm 4)
RESULT_ROOT = "/kaggle/working/results"

MODELS = [
    "cnn_gru",
    "bimamba",
]

EXPERIMENTS = [               # bỏ comment thí nghiệm cần chạy
    "esp_esp",                # 1: ESP32 -> ESP32
    # "asus_asus",            # 2: ASUS (4 ăng-ten) -> ASUS
    # "asus0_esp",            # 3: ASUS (ăng-ten 0) -> ESP32
    # "xrf55",                # 4: XRF55 -> XRF55
    # "esp_asus0",            # 5: ESP32 -> ASUS (ăng-ten 0)
]

def data_root_of(exp):
    return XRF55_ROOT if exp == "xrf55" else ROOM2_ROOT

for exp in EXPERIMENTS:       # kiểm tra đường dẫn: room 2 phải thấy room_02, XRF55 phải thấy file .npy
    items = sorted(os.listdir(data_root_of(exp)))
    print(exp, "->", data_root_of(exp), ":", items[:3], "...", len(items), "mục")

# ===== Cell 4: chạy =====
for exp in EXPERIMENTS:
    data_root = data_root_of(exp)
    for model in MODELS:
        print(f"\n{'#' * 20} exp={exp} / model={model} {'#' * 20}")
        !python main.py --exp {exp} --model {model} --data_root {data_root} --result_root {RESULT_ROOT}

# ===== Cell 5: nén kết quả để tải về =====
ZIP = "/kaggle/working/results_v2_" + "_".join(EXPERIMENTS) + ".zip"
