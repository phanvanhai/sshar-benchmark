import os, subprocess
os.environ["PYDEVD_DISABLE_FILE_VALIDATION"] = "1"

    subprocess.run(f"git fetch origin && git checkout {BRANCH} && git pull origin {BRANCH}",
                   shell=True, check=True, cwd=REPO)

print(subprocess.run("git branch --show-current && git log --oneline -1",
                     shell=True, capture_output=True, text=True).stdout)   # phải thấy: v2

# ===== Cell 3: cấu hình =====
ROOM_ROOTS = {                # thư mục CHỨA room_0X (thí nghiệm 1, 2, 3, 5)
    "room_01": "/kaggle/input/datasets/phanvanhai/cmr-har-room1",  # <-- sửa: thư mục chứa room_01
    "room_02": "/kaggle/input/datasets/phanvanhai/cmr-har-room2",  # <-- sửa: thư mục chứa room_02
}
XRF55_ROOT  = "/kaggle/input/datasets/phanvanhai/xrf55-8act/XRF55_8act_dataset" # <-- sửa: thư mục chứa file .npy XRF55 (thí nghiệm 4)
RESULT_ROOT = "/kaggle/working/results"

ROOMS = [                     # phòng cần chạy (XRF55 không phụ thuộc phòng)
    # "room_01",
    "room_02",
]

MODELS = [
    "cnn_gru",
    "bimamba",
]

EXPERIMENTS = [               # bỏ comment thí nghiệm cần chạy
    "esp_esp",                # 1: ESP32 -> ESP32
    # "asus_asus",            # 2: ASUS (4 ăng-ten) -> ASUS
    # "asus0_esp",            # 3: ASUS (ăng-ten 0) -> ESP32
    # "xrf55",                # 4: XRF55 -> XRF55 (chạy 1 lần, bỏ qua ROOMS)
    # "esp_asus0",            # 5: ESP32 -> ASUS (ăng-ten 0)
]

RUNS = []                     # (phòng, thí nghiệm, data_root)
for exp in EXPERIMENTS:
    if exp == "xrf55":
        RUNS.append((None, exp, XRF55_ROOT))
    else:
        RUNS += [(room, exp, ROOM_ROOTS[room]) for room in ROOMS]

for room, exp, root in RUNS:  # kiểm tra đường dẫn: phải thấy room_0X, hoặc file .npy với XRF55
    items = sorted(os.listdir(root))
    print(room, exp, "->", root, ":", items[:3], "...", len(items), "mục")

# ===== Cell 4: chạy =====
for room, exp, root in RUNS:
    room_arg = "" if room is None else f"--room {room}"
    for model in MODELS:
        print(f"\n{'#' * 20} room={room} / exp={exp} / model={model} {'#' * 20}")

ZIP = "/kaggle/working/results_v2_" + "_".join(ROOMS + EXPERIMENTS) + ".zip"
