# ===== Cell 3: cấu hình =====
FOLD    = 1                   # <-- sửa: 1..4
SESSION = "S1"                # <-- sửa: S1 / S2 / S3

ROOM_ROOTS = {                # thư mục CHỨA room_0X
    "room_01": "/kaggle/input/datasets/phanvanhai/cmr-har-room1",
    "room_02": "/kaggle/input/datasets/phanvanhai/cmr-har-room2",
}
XRF55_ROOT  = "/kaggle/input/datasets/phanvanhai/xrf55-8act/XRF55_8act_dataset"
RESULT_ROOT = "/kaggle/working/results"
MAX_HOURS   = 9.0             # không bắt đầu lần chạy mới nếu phiên đã chạy quá MAX_HOURS giờ

SESSIONS = {                  # (phòng, thí nghiệm, mô hình); XRF55: phòng = None
    "S1": [("room_01", "esp_esp", "cnn_gru"), ("room_01", "esp_esp", "bimamba"),
           ("room_02", "esp_esp", "cnn_gru"), ("room_02", "esp_esp", "bimamba")],
    "S2": [("room_01", "asus_asus", "cnn_gru"), ("room_01", "asus_asus", "bimamba"),
           (None, "xrf55", "bimamba")],
    "S3": [("room_02", "asus_asus", "cnn_gru"), ("room_02", "asus_asus", "bimamba"),
           (None, "xrf55", "cnn_gru")],
}
assert FOLD in (1, 2, 3, 4, 5) and SESSION in SESSIONS
RUNS = SESSIONS[SESSION]

def run_name(room, exp, model):
    base = f"{exp}_{model}" if room is None else f"{room}_{exp}_{model}"
    return f"{base}_fold{FOLD}"

for room, exp, model in RUNS:  # kiểm tra đường dẫn: phải thấy room_0X, hoặc file .npy với XRF55
    root = XRF55_ROOT if room is None else ROOM_ROOTS[room]
    items = sorted(os.listdir(root))
    print(run_name(room, exp, model), "->", root, ":", items[:3], "...", len(items), "mục")

# ===== Cell 4: chạy (nén kết quả sau mỗi lần chạy) =====
ZIP = f"/kaggle/working/results_v3_fold{FOLD}_{SESSION}.zip"
for room, exp, model in RUNS:
    name = run_name(room, exp, model)
    hours = (time.time() - SESSION_START) / 3600
    if os.path.exists(f"{RESULT_ROOT}/{name}/best_val/metrics.json"):
        print(f"\n[bỏ qua] {name}: đã có kết quả")
        continue
    if hours > MAX_HOURS:
        print(f"\n[dừng] phiên đã chạy {hours:.1f} h > {MAX_HOURS} h, chưa chạy: {name}")
        break
    root = XRF55_ROOT if room is None else ROOM_ROOTS[room]
    room_arg = "" if room is None else f"--room {room}"
    print(f"\n{'#' * 20} {name}  (phiên đã chạy {hours:.1f} h) {'#' * 20}")
    !python main.py --exp {exp} {room_arg} --fold {FOLD} --model {model} --data_root {root} --result_root {RESULT_ROOT}
    !cd /kaggle/working && rm -f {ZIP} && zip -qr {ZIP} results
    print("Saved:", ZIP)

print(f"\nXong phiên fold{FOLD} {SESSION} sau {(time.time() - SESSION_START) / 3600:.1f} h")
