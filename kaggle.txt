# Train/val: room 1 (fold k: val = nhóm lần lặp k, train = 4 nhóm còn lại); test: TOÀN BỘ room 2.
# Mỗi phiên chỉ cần sửa 2 dòng ở Cell 3: ACCOUNT (TK1..TK5) và PART (phiên thứ mấy của tài khoản đó);
# fold và nhóm lần chạy được lấy tự động từ bảng PLAN.
#   S1: ASUS CNN-GRU + ESP CNN-GRU      ~6,1 h / fold
#   S2: ASUS BiMamba + ESP BiMamba      ~4,5 h / fold
#   SX: XRF55 CNN-GRU + XRF55 BiMamba   ~2,4 h / fold
# Tổng ~65 h GPU (ước tính). Phân công 5 tài khoản:
#   TK1..TK4 (TKk): PART 1 = fold k S1, PART 2 = fold k S2                  (~10,6 h mỗi tài khoản)
#   TK5           : PART 1 = fold 5 S1, PART 2 = fold 5 S2,
#                   PART 3 = SX fold 1,2,3, PART 4 = SX fold 4,5             (~22,6 h)
# Chạy S1 trước ở mọi tài khoản và xem thời gian thực tế trước khi chạy S2.

# ===== Cell 3: cấu hình =====
ACCOUNT = "TK1"               # <-- sửa: TK1 / TK2 / TK3 / TK4 / TK5
PART    = 1                   # <-- sửa: phiên thứ mấy của tài khoản (TK1-TK4: 1..2, TK5: 1..4)

PLAN = {                      # tài khoản -> các phiên theo thứ tự: (danh sách fold, nhóm lần chạy)
    "TK1": [([1], "S1"), ([1], "S2")],
    "TK2": [([2], "S1"), ([2], "S2")],
    "TK3": [([3], "S1"), ([3], "S2")],
    "TK4": [([4], "S1"), ([4], "S2")],
    "TK5": [([5], "S1"), ([5], "S2"), ([1, 2, 3], "SX"), ([4, 5], "SX")],
}
assert ACCOUNT in PLAN and 1 <= PART <= len(PLAN[ACCOUNT]), f"{ACCOUNT} có {len(PLAN.get(ACCOUNT, []))} phiên"
FOLDS, SESSION = PLAN[ACCOUNT][PART - 1]
print(f"{ACCOUNT} phiên {PART}/{len(PLAN[ACCOUNT])}: folds={FOLDS}, {SESSION}")

SRC_ROOM_ROOT  = "/kaggle/input/datasets/phanvanhai/cmr-har-room1"   # thư mục CHỨA room_01 (nguồn)
TGT_ROOM_ROOT  = "/kaggle/input/datasets/phanvanhai/cmr-har-room2"   # thư mục CHỨA room_02 (đích)
SRC_XRF55_ROOT = "/kaggle/input/datasets/phanvanhai/xrf55-8act/XRF55_8act_dataset"  # file .npy XRF55 room1
TGT_XRF55_ROOT = "/kaggle/input/datasets/phanvanhai/xrf55-room2"    # thư mục file .npy XRF55 room2
RESULT_ROOT = "/kaggle/working/results"
MAX_HOURS   = 7.5             # không bắt đầu lần chạy mới nếu phiên đã chạy quá MAX_HOURS giờ

SESSIONS = {                  # (thí nghiệm, mô hình)
    "S1": [("cross_env_asus", "cnn_gru"), ("cross_env_esp", "cnn_gru")],
    "S2": [("cross_env_asus", "bimamba"), ("cross_env_esp", "bimamba")],
    "SX": [("cross_env_xrf55", "cnn_gru"), ("cross_env_xrf55", "bimamba")],
}
assert SESSION in SESSIONS and FOLDS and all(k in (1, 2, 3, 4, 5) for k in FOLDS)
RUNS = [(fold, exp, model) for fold in FOLDS for exp, model in SESSIONS[SESSION]]

def roots(exp):
    return (SRC_XRF55_ROOT, TGT_XRF55_ROOT) if exp == "cross_env_xrf55" else (SRC_ROOM_ROOT, TGT_ROOM_ROOT)

def run_name(fold, exp, model):
    return f"r1_to_r2_{exp.replace('cross_env_', '')}_{model}_fold{fold}"

for exp in sorted({exp for _, exp, _ in RUNS}):   # kiểm tra đường dẫn: phải thấy room_0X, hoặc file .npy với XRF55
    for root in roots(exp):
        items = sorted(os.listdir(root))
        print(exp, "->", root, ":", items[:3], "...", len(items), "mục")
print("Sẽ chạy:", [run_name(*r) for r in RUNS])

# ===== Cell 4: chạy (nén kết quả sau mỗi lần chạy) =====
ZIP = f"/kaggle/working/results_cross_env_{ACCOUNT}_part{PART}_fold{''.join(map(str, FOLDS))}_{SESSION}.zip"
for fold, exp, model in RUNS:
    name = run_name(fold, exp, model)
    hours = (time.time() - SESSION_START) / 3600
    if os.path.exists(f"{RESULT_ROOT}/{name}/best_val/metrics.json"):
        print(f"\n[bỏ qua] {name}: đã có kết quả")
        continue
    if hours > MAX_HOURS:
        print(f"\n[dừng] phiên đã chạy {hours:.1f} h > {MAX_HOURS} h, chưa chạy: {name}")
        break
    src, tgt = roots(exp)
    print(f"\n{'#' * 20} {name}  (phiên đã chạy {hours:.1f} h) {'#' * 20}")
    t_run = time.time()
    !python main.py --exp {exp} --fold {fold} --model {model} --data_root {src} --tgt_root {tgt} --result_root {RESULT_ROOT}
    print(f"[thời gian] {name}: {(time.time() - t_run) / 3600:.2f} h")
    !cd /kaggle/working && rm -f {ZIP} && zip -qr {ZIP} results
    print("Saved:", ZIP)

print(f"\nXong {ACCOUNT} phiên {PART}: folds={FOLDS} {SESSION} sau {(time.time() - SESSION_START) / 3600:.1f} h")
